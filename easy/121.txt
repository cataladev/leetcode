public class Solution {
    public int MaxProfit(int[] prices) {
        int l = 0;
        int r = 1;
        int max = 0;
        while(r<prices.Length){
            int curr = prices[r] - prices[l];
            if(curr > max){
                max = curr;
            }
            else if(prices[l] > prices[r]){
                l = r;
            }
            r += 1;
        }
        return max;
    }
}
