public class Solution {
    public int CherryPickup(int[][] grid) {
        int n = grid.Length;
        const int NEG = -1_000_000_000;

        int[,] dp = new int[n,n];
        for (int i = 0; i < n; i++){
            for (int j = 0; j < n; j++){
                dp[i, j] = NEG;
            }
        }
        dp[0,0] = grid[0][0] == -1 ? NEG : grid[0][0];

        for (int k = 1; k <= 2 * (n-1); k++) {
            int[,] next = new int[n,n];
            for(int i = 0; i < n; i++){
                for(int j = 0; j < n; j++){
                    next[i,j] = NEG;
                }
            }
            int rMin = Math.Max(0, k - (n-1));
            int rMax = Math.Min(n - 1, k);

            for (int r1 = rMin; r1 <= rMax; r1++) {
                int c1 = k - r1;
                if (c1 < 0 || c1 >= n || grid[r1][c1] == -1){
                    continue;
                }

                for(int r2 = rMin; r2 <= rMax; r2++){
                                        int c2 = k - r2;
                    if (c2 < 0 || c2 >= n || grid[r2][c2] == -1) continue;

                    int bestPrev = dp[r1, r2]; // both came from left (both moved right previously)
                    if (r1 > 0) bestPrev = Math.Max(bestPrev, dp[r1 - 1, r2]);        // r1 from up
                    if (r2 > 0) bestPrev = Math.Max(bestPrev, dp[r1, r2 - 1]);        // r2 from up
                    if (r1 > 0 && r2 > 0) bestPrev = Math.Max(bestPrev, dp[r1 - 1, r2 - 1]); // both from up
                    if (bestPrev <= NEG / 2) continue;

                    int gain = grid[r1][c1];
                    if (r1 != r2 || c1 != c2) gain += grid[r2][c2];

                    next[r1, r2] = Math.Max(next[r1, r2], bestPrev + gain);
                }
            }
            dp = next;
        }

        int ans = dp[n - 1, n - 1];
        return ans < 0 ? 0 : ans;
    }
}
